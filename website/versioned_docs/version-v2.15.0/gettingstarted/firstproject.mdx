func IsAdvertisement(text string) bool {

	adKeywords := []string{
		"광고",
		"AD",
		"Sponsored",
	}

	for _, keyword := range adKeywords {
		if strings.Contains(
			strings.ToLower(text),
			strings.ToLower(keyword),
		) {
			return true
		}
	}

	return false

func ExtractTags(title string) []string {

	title = strings.ReplaceAll(title, "[", " ")
	title = strings.ReplaceAll(title, "]", " ")
	title = strings.ReplaceAll(title, "(", " ")
	title = strings.ReplaceAll(title, ")", " ")
	title = strings.ReplaceAll(title, "/", " ")

	words := strings.Fields(title)

	seen := make(map[string]bool)
	tags := []string{}

	for _, word := range words {

		if len([]rune(word)) < 2 {
			continue
		}

		if seen[word] {
			continue
		}

		seen[word] = true
		tags = append(tags, word)
	}

	return tags
}

<script>
    import { SearchNaver } from "../wailsjs/go/main/App";

    let keyword = "";
    let loading = false;
    let result = null;
    let error = "";

    async function search() {

        if (!keyword.trim()) {
            return;
        }

        loading = true;
        error = "";

        try {

            result = await SearchNaver(keyword);

        } catch (e) {

            console.error(e);
            error = "검색 중 오류가 발생했습니다.";

        } finally {

            loading = false;
        }
    }
</script>
